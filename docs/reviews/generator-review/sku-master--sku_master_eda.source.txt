def actual_vs_target_table(counts: pd.Series, targets: dict, name: str) -> pd.DataFrame:
    rows = []
    for key, target in targets.items():
        got = int(counts.get(key, 0))
        rows.append({"dimension": name, "value": key, "actual": got, "target": target, "gap": got - target})
    return pd.DataFrame(rows)


def show_fig(fig):
    """Display Plotly figure in Jupyter."""
    return fig.show()


def count_bar(series: pd.Series, title: str) -> pd.DataFrame:
    counts = series.value_counts().reset_index()
    counts.columns = [series.name or "category", "count"]
    fig = px.bar(counts, x=counts.columns[0], y="count", title=title, text="count")
    fig.update_traces(textposition="outside")
    show_fig(fig)
    return counts

# NOTEBOOK CELL 23 (zero-based)
invalid_color_rows: list[str] = []
invalid_size_rows: list[str] = []
violations: list[dict] = []

for row in df.itertuples(index=False):
    st = STYLES.get(row.style_id)
    if st is None:
        violations.append({"rule": "unknown_style", "detail": row.style_id})
        continue
    if row.color not in st["allowed_colors"]:
        invalid_color_rows.append(row.sku_id)
    if row.size not in st["allowed_sizes"]:
        invalid_size_rows.append(row.sku_id)
    if row.unit_price != st["unit_price"] or row.price_band != st["price_band"]:
        violations.append({"rule": "price", "detail": row.sku_id})
    if band_from_price(row.unit_price) != row.price_band:
        violations.append({"rule": "band_derive", "detail": row.sku_id})
    if row.return_window_days != CONFIG["return_window_by_category"][st["category"]]:
        violations.append({"rule": "return_window", "detail": row.sku_id})
    exp_occ = row.style_id in CONFIG["occasion_styles"]
    if row.is_occasion_wear != exp_occ:
        violations.append({"rule": "occasion", "detail": row.sku_id})

dup_var = df[df.duplicated(subset=["style_id", "color", "size"], keep=False)]

def realism_checks(frame: pd.DataFrame) -> pd.DataFrame:
    invalid_color = 0
    invalid_size = 0
    for row in frame.itertuples(index=False):
        st = STYLES[row.style_id]
        if row.color not in st["allowed_colors"]:
            invalid_color += 1
        if row.size not in st["allowed_sizes"]:
            invalid_size += 1

    per_style = frame.groupby("style_id").agg(
        n_price=("unit_price", "nunique"),
        n_band=("price_band", "nunique"),
        n_launch=("launch_date", "nunique"),
        n_occ=("is_occasion_wear", "nunique"),
        n_active=("active_flag", "nunique"),
    )
    inconsistent_styles = int((per_style > 1).any(axis=1).sum())

    rows = [
        ("Style -> allowed colours/sizes", invalid_color == 0 and invalid_size == 0),
        ("Style -> price band (constant per style)", per_style["n_band"].max() == 1),
        ("Style -> single list price", per_style["n_price"].max() == 1),
        ("Size/colour -> price (independent)", frame.groupby(["style_id", "color"])["unit_price"].nunique().max() == 1),
        ("Band <-> numeric price (v2)", (frame["price_band"] == frame["derived_band"]).all()),
        ("Style -> single launch / occasion / active", inconsistent_styles == 0),
        ("Category -> return window", frame.groupby("category")["return_window_days"].nunique().max() == 1 and frame["category"].nunique() >= 5),
    ]
    return pd.DataFrame(rows, columns=["relationship", "realistic"])


realism_scorecard = realism_checks(df)

def iqr_outliers(series: pd.Series) -> pd.Series:
    q1, q3 = series.quantile(0.25), series.quantile(0.75)
    iqr = q3 - q1
    return (series < q1 - 1.5 * iqr) | (series > q3 + 1.5 * iqr)


global_out = df[iqr_outliers(df["unit_price"])]
