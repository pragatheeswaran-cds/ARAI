# NOTEBOOK CELL 1 (zero-based)
from pathlib import Path

import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.cluster import DBSCAN
from sklearn.decomposition import PCA
from sklearn.metrics import (
    adjusted_rand_score,
    normalized_mutual_info_score,
    silhouette_score,
)
from sklearn.neighbors import NearestNeighbors
from sklearn.preprocessing import StandardScaler

sns.set_theme(style="whitegrid", context="notebook")
plt.rcParams["figure.figsize"] = (10, 4.5)
plt.rcParams["axes.titlesize"] = 12

ROOT = Path("..").resolve()
if not (ROOT / "output" / "customer_master.csv").exists():
    ROOT = Path.cwd()
    if not (ROOT / "output" / "customer_master.csv").exists():
        ROOT = Path(r"C:\Users\Abhinandan\Projects\customer_master_local")

CSV = ROOT / "output" / "customer_master.csv"
print("project root:", ROOT)
print("csv:", CSV)

# NOTEBOOK CELL 3 (zero-based)
df = pd.read_csv(CSV)
print(f"rows={len(df):,}  cols={len(df.columns)}")
print("all columns:", list(df.columns))

LABEL_COLS = ["customer_segment", "account_status"]
ID_COLS = ["customer_id", "signup_date"]
FEATURE_COLS = [
    "tenure_days",
    "actual_orders",
    "avg_days_between_orders",
    "festive_order_ratio",
    "return_rate",
    "shared_device_count",
]

missing = [c for c in LABEL_COLS + ID_COLS + FEATURE_COLS if c not in df.columns]
if missing:
    raise ValueError(f"missing columns: {missing}")

cluster_df = df.copy()
cluster_df["avg_days_between_orders"] = cluster_df["avg_days_between_orders"].fillna(
    cluster_df["tenure_days"]
)

X_raw = cluster_df[FEATURE_COLS].copy()
X_raw["actual_orders"] = np.log1p(X_raw["actual_orders"].astype(float))
X_raw["avg_days_between_orders"] = np.log1p(X_raw["avg_days_between_orders"].astype(float))

scaler = StandardScaler()
X = scaler.fit_transform(X_raw)

hidden_segment = cluster_df["customer_segment"].astype(str)
hidden_status = cluster_df["account_status"].astype(str)

print("dropped labels :", LABEL_COLS)
print("dropped ids    :", ID_COLS)
print("features used  :", FEATURE_COLS)
print("X shape        :", X.shape)
print("NaNs in X      :", int(np.isnan(X).sum()))
print("hidden segment :", hidden_segment.value_counts().to_dict())
print("hidden status  :", hidden_status.value_counts().to_dict())
cluster_df[FEATURE_COLS].describe().round(2)

# NOTEBOOK CELL 5 (zero-based)
min_samples = 2 * X.shape[1]
nn = NearestNeighbors(n_neighbors=min_samples)
nn.fit(X)
dists, _ = nn.kneighbors(X)
kth = np.sort(dists[:, -1])

fig, ax = plt.subplots(figsize=(8, 4))
ax.plot(kth)
ax.set_title(f"{min_samples}-distance plot (look for the elbow → eps)")
ax.set_xlabel("Points sorted by distance")
ax.set_ylabel(f"Distance to {min_samples}th neighbor")
plt.tight_layout()
plt.show()

print("min_samples:", min_samples)
print("k-distance quantiles:")
print(pd.Series(kth).quantile([0.50, 0.75, 0.85, 0.90, 0.95, 0.97, 0.99]).round(3).to_string())

# NOTEBOOK CELL 7 (zero-based)
eps_candidates = np.unique(np.round(np.quantile(kth, [0.85, 0.90, 0.95, 0.97, 0.99]), 2))
rows = []
db_runs = {}
for eps in eps_candidates:
    db = DBSCAN(eps=float(eps), min_samples=min_samples)
    pred = db.fit_predict(X)
    db_runs[float(eps)] = pred
    n_clusters = len(set(pred) - {-1})
    n_noise = int((pred == -1).sum())
    labeled = pred != -1
    if n_clusters >= 2 and labeled.sum() > n_clusters:
        sil = silhouette_score(X[labeled], pred[labeled])
    else:
        sil = np.nan
    rows.append(
        {
            "eps": float(eps),
            "n_clusters": n_clusters,
            "n_noise": n_noise,
            "noise_pct": round(100 * n_noise / len(pred), 1),
            "silhouette_non_noise": None if np.isnan(sil) else round(float(sil), 3),
            "ARI_segment": round(adjusted_rand_score(hidden_segment, pred), 3),
            "ARI_status": round(adjusted_rand_score(hidden_status, pred), 3),
        }
    )

db_scan = pd.DataFrame(rows)
display(db_scan)

ok = db_scan[(db_scan["n_clusters"] >= 2) & (db_scan["noise_pct"] < 40)]
if len(ok):
    pick = ok.sort_values(["ARI_segment", "silhouette_non_noise"], ascending=False).iloc[0]
else:
    pick = db_scan.iloc[0]
eps_pick = float(pick["eps"])
db_labels = db_runs[eps_pick]
print("picked eps:", eps_pick)
print(pick.to_dict())

# NOTEBOOK CELL 9 (zero-based)
def cluster_quality(pred, title):
    print(title)
    print("  sizes:", pd.Series(pred).value_counts().sort_index().to_dict())
    print("  ARI vs customer_segment :", round(adjusted_rand_score(hidden_segment, pred), 3))
    print("  NMI vs customer_segment :", round(normalized_mutual_info_score(hidden_segment, pred), 3))
    print("  ARI vs account_status   :", round(adjusted_rand_score(hidden_status, pred), 3))
    print("  NMI vs account_status   :", round(normalized_mutual_info_score(hidden_status, pred), 3))
    print()
    print("row-normalized: segment vs cluster")
    display(pd.crosstab(hidden_segment, pred, normalize="index").round(2))
    print("row-normalized: account_status vs cluster")
    display(pd.crosstab(hidden_status, pred, normalize="index").round(2))


cluster_quality(db_labels, f"DBSCAN eps={eps_pick}, min_samples={min_samples}")

profile = (
    cluster_df.assign(dbscan=db_labels)
    .groupby("dbscan")[FEATURE_COLS]
    .median()
)
print("cluster medians (original units; -1 = noise)")
display(profile.round(2))

n_noise = int((db_labels == -1).sum())
print(f"noise points: {n_noise:,} ({100 * n_noise / len(db_labels):.1f}%)")
if n_noise:
    print("noise by hidden segment:")
    display(hidden_segment[db_labels == -1].value_counts())
    print("noise by hidden account_status:")
    display(hidden_status[db_labels == -1].value_counts())

# NOTEBOOK CELL 11 (zero-based)
pca = PCA(n_components=2, random_state=42)
xy = pca.fit_transform(X)
print(
    "PCA explained variance:",
    np.round(pca.explained_variance_ratio_, 3),
    "sum",
    round(pca.explained_variance_ratio_.sum(), 3),
)

viz = pd.DataFrame(
    {
        "pc1": xy[:, 0],
        "pc2": xy[:, 1],
        "dbscan": np.where(db_labels < 0, "noise", db_labels.astype(str)),
        "segment": hidden_segment,
        "status": hidden_status,
    }
)

fig, axes = plt.subplots(1, 3, figsize=(16, 4.8))
plots = [
    (axes[0], "dbscan", f"DBSCAN (eps={eps_pick})"),
    (axes[1], "segment", "Hidden customer_segment"),
    (axes[2], "status", "Hidden account_status"),
]
for ax, col, title in plots:
    sns.scatterplot(data=viz, x="pc1", y="pc2", hue=col, s=12, alpha=0.55, ax=ax, linewidth=0)
    ax.set_title(title)
    ax.legend(markerscale=1.4, fontsize=8, loc="best")
plt.tight_layout()
plt.show()

# NOTEBOOK CELL 13 (zero-based)
ari_seg = adjusted_rand_score(hidden_segment, db_labels)
ari_st = adjusted_rand_score(hidden_status, db_labels)
n_noise = int((db_labels == -1).sum())
n_clusters = len(set(db_labels) - {-1})
fraud_n = int((hidden_segment == "Fraud Proxy").sum())
fraud_noise = int(((db_labels == -1) & (hidden_segment == "Fraud Proxy")).sum())

print("=== DBSCAN realism checklist ===")
print(f"eps={eps_pick}  min_samples={min_samples}  clusters={n_clusters}")
print(f"ARI(segment)={ari_seg:.3f}  ARI(status)={ari_st:.3f}")
print(f"noise={n_noise} ({100 * n_noise / len(db_labels):.1f}%)")
if fraud_n:
    print(f"Fraud Proxy in noise: {fraud_noise}/{fraud_n} ({100 * fraud_noise / fraud_n:.1f}%)")
if n_noise:
    print("noise composition (share of noise points):")
    print(hidden_segment[db_labels == -1].value_counts(normalize=True).round(3).to_string())

print()
if ari_seg >= 0.85:
    print("segment recovery is very high → personas are still too boxy; add more overlap in generation.")
elif ari_seg >= 0.25:
    print("segment recovery is moderate → density sees structure without memorizing labels. Good sign.")
else:
    print("segment recovery is weak → overlapping density, or personas do not show up as separate blobs.")

if ari_st < 0.15:
    print("account_status is not a density cluster → expected, and realistic.")
else:
    print("account_status is aligning with clusters → status may be too tightly tied to a persona box.")

noise_pct = 100 * n_noise / len(db_labels)
if n_noise == 0:
    print("zero noise → either eps is too large or there is no outlier tail.")
elif noise_pct > 40:
    print("noise is huge → eps is too small, or the table is too diffuse to form cores.")
elif fraud_n and fraud_noise / fraud_n >= 0.2:
    print("Fraud Proxy is enriched in noise → expected for a rare, multi-pattern class.")
else:
    print("some noise exists, but Fraud Proxy is not especially isolated.")
